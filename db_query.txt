        (5, 'Premium Collection'),

numbered_products AS (
    SELECT
        c.id AS category_id,
        c.name AS category_name,
        v.variant_num,
        v.variant_name,
        ROW_NUMBER() OVER (
            ORDER BY c.id, v.variant_num
        ) AS product_num
    FROM categories c
    CROSS JOIN variants v
),
product_data AS (
    SELECT
        p.*,
        i.urls[
            ((p.product_num - 1) % array_length(i.urls, 1)) + 1
        ] AS image1,
        i.urls[
            (p.product_num % array_length(i.urls, 1)) + 1
        ] AS image2
    FROM numbered_products p
    CROSS JOIN image_sources i
)

    ((product_num * 3) % 21)::numeric(5,2),
    10 + ((product_num * 7) % 91),
    image1 || ',' || image2,
    CURRENT_TIMESTAMP,
    category_id
FROM product_data
